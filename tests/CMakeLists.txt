cmake_minimum_required(VERSION 3.20)
project(puls_tests CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
enable_testing()

add_library(puls_engine STATIC ../engine/Sequencer.cpp)
target_include_directories(puls_engine PUBLIC ../engine)
target_compile_options(puls_engine PRIVATE -Wall -Wextra -Wno-unused-parameter)

add_executable(engine_tests EngineTests.cpp)
target_link_libraries(engine_tests puls_engine)
target_compile_options(engine_tests PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_test(NAME engine COMMAND engine_tests)

add_library(puls_app STATIC ../state/Serialization.cpp ../bridge/Bridge.cpp)
target_include_directories(puls_app PUBLIC ../model ../state ../bridge ../third_party)
target_link_libraries(puls_app PUBLIC puls_engine)
target_compile_options(puls_app PRIVATE -Wall -Wextra -Wno-unused-parameter)

add_executable(bridge_tests BridgeTests.cpp)
target_link_libraries(bridge_tests puls_app)
target_compile_options(bridge_tests PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_test(NAME bridge COMMAND bridge_tests)

add_library(puls_gui STATIC ../gui/EditorLayout.cpp ../gui/Commands.cpp ../gui/Menu.cpp ../gui/Selection.cpp ../gui/Help.cpp)
target_include_directories(puls_gui PUBLIC ../gui)
target_link_libraries(puls_gui PUBLIC puls_app)
target_compile_options(puls_gui PRIVATE -Wall -Wextra -Wno-unused-parameter)

add_executable(gui_tests GuiTests.cpp)
target_link_libraries(gui_tests puls_gui)
target_compile_options(gui_tests PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_test(NAME gui COMMAND gui_tests)
